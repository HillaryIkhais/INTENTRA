Install

Shell

TypeScript
// Import the SDK
import { IntentraSdk } from 'intentra-sdk';
// Declare intent
const intent = sdk.setIntent('Buy BTC only. Max $100.');
// Create time-boxed session
const session = sdk.createSession(agentId, intent, {
durationMs: 5 * 60 * 1000, // 5 minutes
maxProposals: 10
});
// Proposal either executes or blocks
const result = await sdk.proposeAndExecute(agentProposal, session.id);
// Revoke at any time
sdk.revokeSession(session.id, 'human', 'Session complete');
